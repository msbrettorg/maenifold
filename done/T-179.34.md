## Intent

New users launching maenifold for the first time see the Home hero but have no daemon connected, no agents configured, and no embedding model downloaded. They need a guided setup flow matching the cowork SetupWizard pattern.

## Reference Implementation

Study: `agency-cowork/ui/src/SetupWizard.jsx` (~2450 lines)

Key patterns to adopt:
- **Layout**: Left sidebar (240px) with step dots + main content area (max-width 640px)
- **State machine**: `step` index + `highWaterStep` for back-navigation
- **Progress events**: IPC stream for long operations (download, extraction)
- **Modal overlays**: Confirmation dialogs overlay the main wizard
- **Launch celebration**: Animated "Done" overlay with glow/shimmer effects
- **Step crossfade**: 0.35s ease animation between steps

## Architecture

### First-Run Detection

```typescript
// useOnboarding.ts
const ONBOARDING_KEY = 'maenifold-onboarding-v1';

interface OnboardingState {
  completed: boolean;
  completedAt?: string;
  version: number;  // Increment to force re-onboarding after major updates
}

export function useOnboarding() {
  const prefsStore = createPrefsStore();  // Reuse from useAppSession pattern
  const isFirstRun = ref(true);
  const step = ref(0);
  const highWaterStep = ref(0);

  async function checkFirstRun(): Promise<boolean> {
    const state = await prefsStore.get<OnboardingState>(ONBOARDING_KEY);
    return !state?.completed || state.version < CURRENT_VERSION;
  }

  async function markComplete() {
    await prefsStore.set(ONBOARDING_KEY, {
      completed: true,
      completedAt: new Date().toISOString(),
      version: CURRENT_VERSION,
    });
    isFirstRun.value = false;
  }

  return { isFirstRun, step, highWaterStep, checkFirstRun, markComplete };
}
```

// In preload or direct fetch
async function checkDaemonHealth(): Promise<DaemonStatus | null> {
  try {
    const res = await fetch('http://localhost:8080/health', { 
      signal: AbortSignal.timeout(3000) 
    });
    if (!res.ok) return null;
    return await res.json() as DaemonStatus;
  } catch {
    return null;
  }
}

### 5. Step navigation state machine

```typescript
const step = ref(0);
const highWaterStep = ref(0);
const canGoBack = computed(() => step.value > 0);
const canGoForward = computed(() => step.value < highWaterStep.value);

function goToStep(target: number) {
  if (target < 0 || target > highWaterStep.value) return;
  step.value = target;
}

function advance() {
  step.value++;
  highWaterStep.value = Math.max(highWaterStep.value, step.value);
}
```
