### File 2: `apps/desktop/src/composables/useMcpPrompts.ts` (NEW)

~80 LOC. Mirror `useMcpResources.ts` shape exactly:

```ts
import type { PromptDescriptor, GetPromptResult } from './mcpTransport';
import { McpCapabilityUnsupportedError } from './useMcpResources';
import { getSharedMcpClient } from './usePipelineMcp';

interface McpPromptsClient {
  listPrompts: () => Promise<PromptDescriptor[]>;
  getPrompt: (name: string, args?: Record<string, unknown>) => Promise<GetPromptResult>;
  onPromptListChanged: (handler: () => void) => () => void;
}

const sharedMcpPrompts = new Map<string, McpPromptsClient>();

export function useMcpPrompts(baseUrl?: string): McpPromptsClient {
  const key = baseUrl ?? '__default__';
  const existing = sharedMcpPrompts.get(key);
  if (existing) return existing;

  const mcp = getSharedMcpClient(baseUrl);
  const handlers = new Set<() => void>();

  mcp.onNotification((notification) => {
    if (notification.method !== 'notifications/prompts/list_changed') return;
    handlers.forEach((handler) => handler());
  });

  mcp.onDisconnect(() => {
    handlers.clear();
  });

  async function ensureConnected(): Promise<void> {
    if (mcp.connected.value) return;
    await mcp.connect();
  }

  async function listPrompts(): Promise<PromptDescriptor[]> {
    await ensureConnected();
    if (!mcp.supportsPrompts()) {
      throw new McpCapabilityUnsupportedError('prompts');
    }
    return mcp.listPrompts();
  }

  async function getPrompt(name: string, args?: Record<string, unknown>): Promise<GetPromptResult> {
    await ensureConnected();
    if (!mcp.supportsPrompts()) {
      throw new McpCapabilityUnsupportedError('prompts');
    }
    return mcp.getPrompt(name, args);
  }

  function onPromptListChanged(handler: () => void): () => void {
    handlers.add(handler);
    return () => { handlers.delete(handler); };
  }

  const client: McpPromptsClient = { listPrompts, getPrompt, onPromptListChanged };
  sharedMcpPrompts.set(key, client);
  return client;
}
```

NOTE: `mcp.supportsPrompts()` is a NEW method on the shared client. Add it to `usePipelineMcp.ts` (or wherever `supportsResourceSubscribe()` lives) by checking `client.getServerCapabilities()?.prompts != null`. ~5 LOC.
